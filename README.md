        // 1. Tự động trượt 2 Banner kích thước bằng nhau
        let currentSlide = 0;
        const totalSlides = 2;
        let slideTimer = null;

        function showSlide(index) {
            currentSlide = (index + totalSlides) % totalSlides;
            const slide0 = document.getElementById('bannerSlide0');
            const slide1 = document.getElementById('bannerSlide1');
            const dot0 = document.getElementById('dot0');
            const dot1 = document.getElementById('dot1');

            if (currentSlide === 0) {
                slide0.classList.remove('hidden');
                slide0.classList.add('block');
                slide1.classList.remove('block');
                slide1.classList.add('hidden');

                dot0.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                dot1.className = 'w-2 h-2 rounded-full bg-white/60 transition-all';
            } else {
                slide1.classList.remove('hidden');
                slide1.classList.add('block');
                slide0.classList.remove('block');
                slide0.classList.add('hidden');

                dot1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                dot0.className = 'w-2 h-2 rounded-full bg-white/60 transition-all';
            }
        }

        function changeSlide(step) {
            showSlide(currentSlide + step);
            resetSlideTimer();
        }

        function goToSlide(index) {
            showSlide(index);
            resetSlideTimer();
        }

        function resetSlideTimer() {
            if (slideTimer) clearInterval(slideTimer);
            slideTimer = setInterval(function () {
                showSlide(currentSlide + 1);
            }, 5000);
        }

        // 2. Bật/tắt menu trên điện thoại
        function toggleMobileMenu() {
            const drawer = document.getElementById('mobileMenuDrawer');
            drawer.classList.toggle('hidden');
        }

        // 3. Chọn dịch vụ điền nhanh xuống Form Báo Giá
        function selectServiceForQuote(serviceTitle) {
            const noteInput = document.getElementById('qNote');
            const formSection = document.getElementById('bao-gia-nhanh');
            if (noteInput) {
                noteInput.value = 'Tôi cần tư vấn & báo giá dịch vụ: ' + serviceTitle;
            }
            if (formSection) {
                formSection.scrollIntoView({ behavior: 'smooth' });
            }
        }

        // 4. Xử lý gửi Form Yêu Cầu & Báo Giá
        function submitQuoteForm(event) {
            event.preventDefault();
            const name = document.getElementById('qName').value.trim();
            const phone = document.getElementById('qPhone').value.trim();
            const car = document.getElementById('qCar').value.trim() || 'Chưa rõ dòng xe';
            const service = document.getElementById('qService').value;
            const note = document.getElementById('qNote').value.trim();

            const successBox = document.getElementById('quoteSuccessBox');
            const successDetail = document.getElementById('quoteSuccessDetail');

            successDetail.textContent = 'Khách hàng: ' + name + ' (' + phone + ') • Xe: ' + car + ' • Hạng mục: ' + service + (note ? ' • Ghi chú: ' + note : '') + '. Garage Ô Tô Trường Sơn sẽ liên hệ lại ngay!';
            successBox.classList.remove('hidden');
        }

        // 5. Lọc danh mục bài viết trang chủ
        function filterPosts(cat, btnElement) {
            const cards = document.querySelectorAll('.post-card');
            cards.forEach(function (card) {
                if (cat === 'all' || card.getAttribute('data-cat') === cat) {
                    card.classList.remove('hidden');
                    card.classList.add('flex');
                } else {
                    card.classList.remove('flex');
                    card.classList.add('hidden');
                }
            });

            const buttons = document.querySelectorAll('.post-tab-btn');
            buttons.forEach(function (btn) {
                btn.className = 'post-tab-btn bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-semibold px-3.5 py-1.5 rounded-full whitespace-nowrap transition';
            });
            if (btnElement) {
                btnElement.className = 'post-tab-btn bg-brand-blue text-white text-xs font-semibold px-3.5 py-1.5 rounded-full whitespace-nowrap transition';
            }
        }

        // 6. Tìm kiếm nhanh dịch vụ / sản phẩm
        function filterProductsBySearch(keyword) {
            const query = (keyword || '').toLowerCase().trim();
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(function (card) {
                const text = card.innerText.toLowerCase();
                if (!query || text.includes(query)) {
                    card.classList.remove('hidden');
                    card.classList.add('flex');
                } else {
                    card.classList.remove('flex');
                    card.classList.add('hidden');
                }
            });
        }

        window.onload = function () {
            resetSlideTimer();
        };
